### SQL Provider Configuration and Notification Templates

Once the Chocolatey packages are complete, run this SQL against your tenant database.

```
DECLARE @StaticAuthProviderId int;
select @StaticAuthProviderId = p.id from core.providertype pt
join core.provider p on pt.id = p.ProviderTypeID and p.Name = 'Static Authentication Provider' and pt.name = 'Authentication'
if(@StaticAuthProviderId is null)
Begin
print('Configuring provider for Static Auth Provider')
insert into core.provider select id, 'Static Authentication Provider', 'Static Authentication Provider' , '', GETUTCDATE() from core.providertype where name = 'Authentication'
End
ELSE
print 'Static Auth Provider is in core.provider table'
select @StaticAuthProviderId = p.id from core.providertype pt
join core.provider p on pt.id = p.ProviderTypeID and p.Name = 'Static Authentication Provider' and pt.name = 'Authentication'
DECLARE @StaticAuthItemId int;
select @StaticAuthItemId = i.id from core.providertype pt
join core.provider p on pt.id = p.ProviderTypeID and p.Name = 'Static Authentication Provider' and pt.name = 'Authentication'
join core.item i on p.id = i.ParentId and i.ItemType = 'Connector'
if(@StaticAuthItemId is null)
Begin
print('Configuring item for Static Auth Provider')
insert into core.item select 'Connector', @StaticAuthProviderId, id , 'Static Auth Provider', GETUTCDATE(), '1.0.0.0', 0, NULL from core.bank
End
ELSE
print 'Static Auth Provider is in core.item table'

DECLARE @AlkamiStepUpProviderId int
select @AlkamiStepUpProviderId = p.id from core.providertype pt
join core.provider p on pt.id = p.ProviderTypeID and p.Name = 'Alkami Step Up' and pt.name = 'StepUp'

DECLARE @SmsStepUpOptionProvider int
select @SmsStepUpOptionProvider = suo.AuthenticatorProviderId from stepup.StepUpOption suo
join stepup.AuthenticationFactor af on af.Name = 'SmsCode'
where suo.AuthenticationFactor = af.ID

IF(@SmsStepUpOptionProvider is null)
BEGIN
print 'adding SmsCode stepup authenticator provider'
insert into stepup.StepUpOption (AuthenticationFactor, AuthenticatorProviderId, [Order],LastModified)
select id, @AlkamiStepUpProviderId, 1, GETUTCDATE() from stepup.AuthenticationFactor where name = 'SmsCode'
END
ELSE
print 'SmsCode authenticator provider already configured'

DECLARE @EmailStepUpOptionProvider int
select @EmailStepUpOptionProvider = suo.AuthenticatorProviderId from stepup.StepUpOption suo
join stepup.AuthenticationFactor af on af.Name = 'EmailCode'
where suo.AuthenticationFactor = af.ID
IF(@EmailStepUpOptionProvider is null)
BEGIN
print 'adding Email stepup authenticator provider'
insert into stepup.StepUpOption (AuthenticationFactor, AuthenticatorProviderId, [Order],LastModified)
select id, @AlkamiStepUpProviderId, 2, GETUTCDATE() from stepup.AuthenticationFactor where name = 'EmailCode'
END
ELSE
print 'EmailCode authenticator provider already configured'

IF((select [Version] from dbo.AlkamiStepup_VersionInfo where [Version] = '20190624077106') is null)
BEGIN
print('Adding default template migration for StepUp')

Declare @BankIdentifier uniqueidentifier
Select @BankIdentifier = BankIdentifier from core.bank

insert into notification.DefaultTemplates
values(@BankIdentifier,'Alkami.MicroServices.StepUp.Alkami.Service.Host', '@model dynamic <p> Verification Code: @Model.Code </p> <p> Expiration Date: @Model.ExpirationDate @Model.TimeZone </p>', '{ "id": "Alkami.MicroServices.StepUp.Data.OneTimePasswordMessageData", "type": "object", "properties": { "Code": { "required": true, "type": [ "string", "null" ] }, "ExpirationDate": { "required": true, "type": [ "string", "null" ] }, "TimeZone": { "required": true, "type": [ "string", "null" ] } } }', 'OneTimePasswordTemplate','SendOneTimePassword', 1, 2, 1033, getutcdate(), 'ORB Financial One Time Password')
insert into notification.DefaultTemplates
values(@BankIdentifier, 'Alkami.MicroServices.StepUp.Alkami.Service.Host', '@model dynamic Verification Code: @Model.Code Expiration Date: @Model.ExpirationDate @Model.TimeZone', '{ "id": "Alkami.MicroServices.StepUp.Data.OneTimePasswordMessageData", "type": "object", "properties": { "Code": { "required": true, "type": [ "string", "null" ] }, "ExpirationDate": { "required": true, "type": [ "string", "null" ] }, "TimeZone": { "required": true, "type": [ "string", "null" ] } } }', 'OneTimePasswordTemplate', 'SendOneTimePassword', 2, 2, 1033, getutcdate(), 'ORB Financial One Time Password')
select * from notification.DefaultTemplates
insert into dbo.AlkamiStepup_VersionInfo values('20190624077106', GETUTCDATE(), 'DEV-77106')
END
ELSE
print 'default template migration for StepUp already exist'
```

### Still not working

update these banksettings to true:
BankSettingName.UseEmailCodeAuthentication
BankSettingName.UseSMSAuthentication

Try reinstalling this service to make sure the migrations have been run. Note: you must have the other microservices started first.

```
choco install alkami.microservices.authentication.workflow.service.host -fy
```
