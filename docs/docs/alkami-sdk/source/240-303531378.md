Mocking one or two data properties is very simple, just pass the items inline when you mount a component and it's done. What happens though when you have all kinds of properties to mock out? And what can you mock in a component?

You can technically mock anything that is on the component instance. The best pattern I have found is to create a function at the top of the test file that returns a valid mock object and mutate it as needed within your specific test case.

#### Source File
NavBar.vue

```
<template>
<header class="app-nav">
<div class="app-nav__breadcrumbs">
<router-link class="app-nav__title" to="/">
<span class="app-nav__appname">Obscurus</span>
</router-link>

<template v-if="teamName">
<span class="app-nav__link app-nav__link--separator"> / </span>
<router-link class="app-nav__link" :to="`/project/${teamName.replace(/\s+/g, '')}`">
<span id="team_name" class="app-nav__link-item">{{ teamName | spacerize }}</span>
</router-link>
</template>

<template v-if="repositoryName">
<span class="app-nav__link app-nav__link--separator"> / </span>
<router-link class="app-nav__link" :to="`/repository/${repositoryName}`">
<span id="repository_name" class="app-nav__link-item">{{ repositoryName }}</span>
</router-link>
</template>

</div>

<div class="app-nav__updated">
<template v-if="shouldShowSyncingLoader">
<loader-circle size="large"></loader-circle>
<div class="app-nav__updated-date">Syncing new data...</div>
</template>

<div v-else class="app-nav__updated-date">Last updated: {{ lastUpdated }}</div>
</div>
</header>
</template>

<script lang="ts">
import Vue from 'vue';
import LoaderCircle from '@/components/global/LoaderCircle.vue';

export default Vue.extend({
name: 'nav-bar',
computed: {
lastUpdated(): string {
const lastUpdateDate = this.$store.state.lastUpdateDate;

if (lastUpdateDate) {
const dateObject = new Date(lastUpdateDate);
return `${dateObject.toDateString()} ${dateObject.getHours()}:${('0' + dateObject.getMinutes()).slice(-2)}`;
}

return '';
},
shouldShowSyncingLoader(): boolean {
return (!this.$store.state.hasFinishedBackgroundSync
&& this.$store.state.hasLoadedInitialData);
},
teamName(): string {
// If there is a route param for the project / team name, use that for the team route
if (this.$route.params.projectName) {
return `${this.$route.params.projectName}`;
}

// Otherwise we are most likely in a repository specific view, so we need to get the information from that.
if (this.$route.params.repositoryName) {
const projectInfo = this.$store.getters.findProjectByRepositoryName(this.$route.params.repositoryName);

// TODO: Figure out how this should be abstracted
return projectInfo.name.split('-')[0];
}

return '';
},
repositoryName(): string {
if (this.$route.params.repositoryName) {
return this.$route.params.repositoryName;
}

return '';
},
},
components: {
LoaderCircle,
},
});
</script>
```

#### Test File
NavBar.spec.ts

```
import { shallowMount } from '@vue/test-utils';
import NavBar from '../../../Scripts/components/global/NavBar.vue';
import LoaderCircle from '../../../Scripts/components/global/LoaderCircle.vue';

// Function that returns a fully valid mock object for the instance
const createValidMockObject = () => {
return {
stubs: ['router-link'], // Stub out the router link component built in to Vue
mocks: {
$store: { // Mock out all store properties referenced in the component
state: {
lastUpdateDate: '2042-01-01',
hasFinishedBackgroundSync: true,
hasLoadedInitialData: true,
},
getters: {
findProjectByRepositoryName: () => { return { name: 'RepositoryName'} },
},
},
$route: { // Also mock out all route parameters that are referenced
params: {
projectName: 'test project',
repositoryName: 'test repository'
},
},
},
filters: {
spacerize: (input: string) => input,
},
};
};

describe('NavBar.vue', () => {
it('renders correctly when store and state variables exist', () => {
// Mount component
const wrapper = shallowMount(NavBar, createValidMockObject());

// Execute tests
const teamName = wrapper.find('#team_name').text();
expect(teamName).toBe('test project');

const repositoryName = wrapper.find('#repository_name').text();
expect(repositoryName).toBe('test repository');

// Does not contain the LoaderCircle component
const updatedElement = wrapper.find('.app-nav__updated');
expect(updatedElement.contains(LoaderCircle)).toBe(false);
});

it('renders correctly when no project name exists', () => {
// Mock component data
const localMock = createValidMockObject();
localMock.mocks.$route.params.projectName = '';

// Mount component
const wrapper = shallowMount(NavBar, localMock);

// Execute tests
const teamName = wrapper.find('#team_name').text();
expect(teamName).toBe('RepositoryName');
});

it('does not render router links if no repository or team name exist', () => {
// Mock component data
const localMock = createValidMockObject();
localMock.mocks.$route.params.projectName = '';
localMock.mocks.$route.params.repositoryName = '';

// Mount component
const wrapper = shallowMount(NavBar, localMock);

// Execute tests
// Neither the team name or repository name router link exist
expect(wrapper.find('#team_name').exists()).toBe(false);
expect(wrapper.find('#repository_name').exists()).toBe(false);
});

it('renders loading indicator if background sync has not finished', () => {
// Mock component data
const localMock = createValidMockObject();
localMock.mocks.$route.params.repositoryName = '';
localMock.mocks.$route.params.projectName = '';
localMock.mocks.$store.state.hasFinishedBackgroundSync = false;

// Mount component
const wrapper = shallowMount(NavBar, localMock);

// Execute tests
const updatedElement = wrapper.find('.app-nav__updated');
expect(updatedElement.contains(LoaderCircle)).toBe(true);
});
});
```
