When the Vue application instance is first created, it is common to need to fetch additional data for the page to be rendered. The easiest pattern to follow to accomplish this is to initialize in the setup method in the composition api. Other methods to utilize may be the Lifecycle Hooks (https://vuejs.org/guide/essentials/lifecycle.html#lifecycle-hooks) for a component.

app.ts

```
import { createPinia } from 'pinia';
import { createApp } from 'vue'; // Imports Vue from node_modules for TypeScript support. Will later be stripped out by Webpack and will need to be included as a script tag in the index.cshtml file.
import App from './App.vue'; // Imports main Vue component to be used for the application.

void (async () => {
const pinia = createPinia()

const app = createApp(App);
app.use(pinia);

app.mount('#app');
})();
```

App.vue

```
<template>
<div class="app-component">
<NavBar />
<router-view name="page" />
</div>
</template>
<script setup lang="ts"> // setup automagically returns and exports all the
import { onMounted } from 'vue';
import { useProjectStore } from './store';

const store = useProjectStore();

store.getInitialData();
</script>
```

store.ts

```
// Interfaces
import { defineStore } from 'pinia'
import * as Api from '../api/requests';
import * as Translator from '../utils/translation';

import { IProject, IProjectAggregate } from '../interfaces/IProject';
import { IExecutionAggregate, IExecution } from '../interfaces/IExecutions';
import { IProjectStore } from '../interfaces/IProjectStore';

export const useProjectStore = defineStore({
state: (): IProjectStore => ({
// Add a variable to state to track the completion of the data fetching
hasLoadedInitialData: false,
// ...
}),
actions: {
// Projects
// =============================================================================
populateProjectAggregateData(payload: IProjectAggregate) {
this.projectAggregate = payload;
},

// Executions
// =============================================================================
populateExecutionAggregateData(payload: IExecutionAggregate) {
this.executionAggregate = payload;
},

// Page Level
// =============================================================================
setHasLoadedInitialData(payload: boolean) {
this.hasLoadedInitialData = payload;
},

async getInitialData() {
// Fetch and transform any needed data for the page to render.
const projectReferences = await Api.getProjectReferences();
const projectExecutions = await Promise.all(projectReferences.map((project) => Api.getProjectExecutions(project.id)));

const translatorCalls = await Promise.all([
Translator.translateProjectRefsToAggregate(projectReferences),
Translator.translateExecutionsToAggregate(projectExecutions),
]);

this.populateProjectAggregateData(translatorCalls[0]);
this.populateExecutionAggregateData(translatorCalls[1]);

// After all mutations have been executed, set the loading state to true
// This should trigger the page to remove the loading component and replace it with the actual page content.
this.setHasLoadedInitialData(true);
},
},
});
```
