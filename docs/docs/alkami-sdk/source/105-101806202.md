At some point in the development of your widget library you might need to send an email. The process outlined here allows an email delivery to be verified by capturing it inside of a local SMTP server.

### Local SMTP Setup

The SMTP server that we will be using for this process is Papercut SMTP. You can download the latest release from https://github.com/ChangemakerStudios/Papercut-SMTP/releases (https://github.com/ChangemakerStudios/Papercut-SMTP/releases).

After installation of Papercut SMTP you must verify that the service is running.

#### Papercut SMTP Options

By default, the options for Papercut SMTP are set as shown in the following screenshot. You cannot run at startup if you are doing extensive testing with the service.

### Update the Notification Provider

Run the attached SQL script to convert the No Op provider, that is installed by default, to an SMTP provider. The script assumes this will be run on DeveloperDynamic database instance, but this can also be run on a stage matched database; you just have to change the Use statement to point to that instance.

Local SMTP Service Update

```
Use DeveloperDynamic
BEGIN TRANSACTION
declare @Debug bit,
@ProviderName nvarchar(100),
@ProviderType nvarchar(100),
@ProviderTypeName nvarchar(100),
@ProviderDescription nvarchar(100),
@ProviderAssemblyInfo nvarchar(254),
@ItemName nvarchar(100),
@IncorrectItemName nvarchar(100),
@ProviderId bigint
set @Debug = 0
set @ProviderType = 'Notification'
set @ProviderName = 'SMTP'
set @ProviderTypeName = 'Notification'
set @ProviderDescription = 'SMTP Provider' -- No Op SMTP Provider
set @ProviderAssemblyInfo = 'Alkami.App.Notification.Provider.Smtp.Provider, Alkami.App.Notification.Provider.Smtp'
-- Alkami.App.Notification.Provider.NoOp.Provider, Alkami.App.Notification.Provider.NoOp
set @ItemName = 'SMTP'
set @IncorrectItemName = 'Email'
set @ProviderId = (SELECT Id
FROM core.Provider
WHERE Name = @ProviderName AND ProviderTypeID = (select ID
from core.ProviderType
where name = @ProviderTypeName))
SELECT @ProviderId
-- SMTP Provider (Update current NoOp provider to SMTP provider)
UPDATE core.Provider
SET Description = @ProviderDescription,
AssemblyInfo = @ProviderAssemblyInfo
WHERE id = @ProviderId and Description like '%No Op%';
-- Do we need to fix the core.Item record
IF EXISTS (SELECT a.Id
FROM core.Item a
where a.Name = @IncorrectItemName and a.ParentId = @ProviderId)
BEGIN
UPDATE core.Item
set Name = @ItemName
where Name = @IncorrectItemName and ParentId = @ProviderId
END
-- SMTP Server
declare @SMTPServer nvarchar(100), @SMTPServerSettingName nvarchar(100)
set @SMTPServer = '127.0.0.1'
set @SMTPServerSettingName = 'SMTP Server'
IF NOT EXISTS (SELECT b.Id
FROM core.Item a
join core.ItemSetting b on b.ItemId = a.Id
where a.Name = @ItemName and a.ParentId = @ProviderId and b.Name = @SMTPServerSettingName
or
a.Name = @IncorrectItemName and a.ParentId = @ProviderId and b.Name = @SMTPServerSettingName)
BEGIN
print('Insert SMTP Server')
INSERT INTO core.ItemSetting
(ItemId, Name, Value, [Version], CreatedUtc)
SELECT Id, @SMTPServerSettingName, @SMTPServer, '1.0.0.0', GETUTCDATE()
from core.Item
where Name = @ItemName and ParentId = @ProviderId;
END
ELSE
BEGIN
print('Update SMTP Server')
UPDATE b
SET Value = @SMTPServer
FROM core.Item a
join core.ItemSetting b on b.ItemId = a.Id
where a.Name = @ItemName and a.ParentId = @ProviderId and b.Name = @SMTPServerSettingName
end
-- SMTP Port
declare @SMTPPort nvarchar(100), @SMTPPortSettingName nvarchar(100)
set @SMTPPort = '25'
set @SMTPPortSettingName = 'SMTP Port'
IF NOT EXISTS (SELECT b.Id
FROM core.Item a
join core.ItemSetting b on b.ItemId = a.Id
where a.Name = @ItemName and a.ParentId = @ProviderId and b.Name = @SMTPPortSettingName)
BEGIN
print('Insert SMTP Port')
INSERT INTO core.ItemSetting
(ItemId, Name, Value, [Version], CreatedUtc)
SELECT Id, @SMTPPortSettingName, @SMTPPort, '1.0.0.0', GETUTCDATE()
from core.Item
where Name = @ItemName and ParentId = @ProviderId
END
ELSE
BEGIN
print('Update SMTP Port')
UPDATE b
SET Value = @SMTPPort
FROM core.Item a
join core.ItemSetting b on b.ItemId = a.Id
where a.Name = @ItemName and a.ParentId = @ProviderId and b.Name = @SMTPPortSettingName
end
-- Use SSL
declare @SMTPUseSSL nvarchar(100), @SMTPUseSSLSettingName nvarchar(100)
set @SMTPUseSSL = 'false'
set @SMTPUseSSLSettingName = 'Use SSL'
-- Don't insert it if it's not found - the default is false
IF EXISTS (SELECT b.Id
FROM core.Item a
join core.ItemSetting b on b.ItemId = a.Id
where a.Name = @ItemName and a.ParentId = @ProviderId and b.Name = @SMTPUseSSLSettingName)
BEGIN
print('Update Use SSL')
UPDATE b
SET Value = @SMTPUseSSL
FROM core.Item a
JOIN core.ItemSetting b on b.ItemId = a.Id
where a.Name = @ItemName and a.ParentId = @ProviderId and b.Name = @SMTPUseSSLSettingName
END
-- Validation
SELECT *
FROM core.Provider
WHERE Name = 'SMTP'
SELECT *
FROM core.Item
WHERE Name = 'SMTP'
SELECT *
FROM core.ItemSetting
WHERE ItemId = 258 AND Name LIKE 'SMTP%'
ROLLBACK
--COMMIT
```

### Send Emails

After all the setup is complete, you can use the following tutorial (https://confluence.alkami.com/sdk/my-first-client-widget-78655984.html) to update an existing widget or create a new one. The following code is based on the SendEmail sample project in the Alkami.Sdk.Samples. This code will accept a POST action from the browser and call the Notification service to send the email.

Controller Code

```
using Alkami.Client.Services.Notification.Repository;
using Alkami.App.Notification.Contracts;
using Alkami.Security.Common.DataContracts;
using Alkami.Client.Messages;
using Training.Client.Widget.SendEmail.Models; // This should be replaced with the namespace of your models
using System.Linq;

[...]

[HttpPost]
public ActionResult Index(TrainingSendEmailModel model)
{
try
{
UserContactEmail emailContact =
CurrentUser.GetUserContact<UserContactEmail>(x => x.IsPrimary).FirstOrDefault()
?? CurrentUser.GetUserContact<UserContactEmail>().FirstOrDefault();

Logger.DebugFormat("[POST] Controller/Index");
if (SendEmail(emailContact, model.EmailSubject, model.EmailBody))
{
ViewData["IsErrorState"] = false;
ViewData["Message"] = "Message Sent";
}
else
{
ViewData["IsErrorState"] = true;
ViewData["Message"] = "Message NOT sent";
}
return View("Index", model);
}
catch (Exception e)
{
Logger.Error("Error [POST] Controller/Index", e);

ViewData["IsErrorState"] = true;
ViewData["Message"] = "Error [POST] Controller/Index " + e.Message;
return View("Error");
}
}

private bool SendEmail(UserContactEmail userContactEmail, string subject, string body)
{
bool isSent = false;

try
{
var result = NotificationRepository.SendToSpecificAddess(
this.CurrentBankIdentifier,
CurrentBankName,
CurrentUserIdentifier,
TransportAgentMedium.Email,
userContactEmail.Email.ToString(),
subject,
body,
true,
true);
if (result == null)
{
throw new InvalidOperationException("SendToUser response returns null.");
}

if (result.Status != Status.Success)
{
string errorMsg = string.Format(
"SendEmail failed to send email message. User Identifier [{0}] Email Address: [{1}] Status [{2}] Error [{3}] ",
new object[]
{
CurrentUserIdentifier.ToString(), userContactEmail.Email.ToString(), result.Status.ToString(),
result.Exception == null ? "" : result.Exception.Message
});
throw new InvalidOperationException(errorMsg);
}
else
{
isSent = true;
}
}
catch (Exception e)
{
Logger.Error("SendEmail failed.", e);
isSent = false;
}

return isSent;
}
```

In this example, the predefined model is altered, and has the appropriate supporting properties for sending an email. In your widget, you would probably create a new model for this.
