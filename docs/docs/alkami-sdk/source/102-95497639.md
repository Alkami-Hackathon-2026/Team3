At Alkami we use the following technologies for unit testing

- NUnit 3: This is our unit testing framework. You can read more about the features and specifics of NUnit by going to their site.
- Moq: This is what we use as our mocking library. It is very easy to get acclimated to and use. You can read more about it here

Now let's get things setup in our development environment so that you will be able to run these tests from the "Test Explorer" in Visual Studio.

The first thing that we need to do is install the NUnit3 Test Adapter extension in Visual Studio following the instructions here (https://docs.nunit.org/articles/vs-test-adapter/Adapter-Installation.html).

After installing NUnit you will need to add the Moq library to your project using the nuget package. The command to install that from the package manager is

```
Install-Package Moq -Version 4.14.5
```

Now that everything is installed we can start adding unit tests to your solution.

#### Service Unit Testing:

Create a new class for unit tests or add your tests to the ServiceTests class and use example code below as a guide. The code file is also attached to this document.

Do to template versioning there might be a difference in the example code and the template, namely the use of OneTimeSetUp annotation in the example code versus SetUp in the template. OneTimeSetUp is more performant and is the one that should be used.

USBFIAccountTests

```
using Alkami.Data.Access;
using Alkami.Data.Validations;
using Alkami.MicroServices.Accounts.Contracts.Requests;

using Alkami.Test;
using Alkami.Test.EntityFrameworkHelpers;
using Moq;
using NUnit.Framework;
using System;
using System.Collections.Generic;
using System.Data.Common;
using System.Linq;
using System.Linq.Expressions;
using System.Security.Claims;
using System.Threading.Tasks;
using Alkami.MicroServices.Accounts.Contracts.Filters_And_Mappers;

namespace USBFI.MicroServices.Accounts.Service.Tests.AccountImplTests
{
// The [TestFixture] annotation is used to inform NUnit that this class contains tests.
[TestFixture]
public class AccountTests
{
// Declare the service object
private AccountServiceImp _accountServiceImpl;

// Declare the Moq objects. We will define the implementations in [OneTimeSetUp]
// These will be typed as the specific objects that they will be mocking.
private Mock<IDataScope> _dataScopeMock;
private Mock<DbConnection> _dbConnectionMock;
private Mock<AccountsContext> _accountContextMock;
private Mock<ISettingsServiceContract> _settingServiceMock;

// Declare any other static data that will be used through the test class
private Guid[] accountIdentifiers =
{
Guid.Parse("9C6B80ED-9242-4D96-8D25-CD7B5ABA8FC5"),
Guid.Parse("A0EE1DCC-51B8-4614-85DC-98CF0B5E7E5E"),
Guid.Parse("B89993ED-B6E9-49DA-AFB9-8635050934A5"),
Guid.Parse("45CB0C9C-5B58-4596-B06E-456377B4A07F"),
Guid.Parse("5864EDD3-2F0F-4FAB-B96D-217311EA75B3"),
Guid.Parse("61519132-6065-473D-9E61-9B0FF75EDB48"),
Guid.Parse("6FCB4E39-5CB4-44CC-9AC0-F9E23D48600C"),
Guid.Parse("15D4EB92-0153-4CC3-B72D-089616B18287"),
};

// This section will define the implementation of the objects to support the unit tests.
// We specifically want to highlight the implementation of the Moq objects.
[OneTimeSetUp]
public void Setup()
{
// Instantiate all of the real objects that will be used needed.
PermissionHelpers.AccountGuidsToUse = accountIdentifiers.ToList();
EntityValidator.AddValidator(new AccountValidator());
_accountServiceImpl = new AccountServiceImp();

// Each moq object will be newed up to the object that it is mocking.
_dbConnectionMock = new Mock<DbConnection>();
// Moq supports a number of method extensions but we will only be using Setup and Returns.
// Setup uses a linq structure to allow you to access properties and methods on the mocked object.
// Returns defines the object returned from the property or method.
_dbConnectionMock.Setup(x => x.ConnectionString).Returns("MockConnectionString");
// In order to use a mocked object in the instantiation of another object you will need to use
// the object property of that object. If this is not used an object reference error will surface.
_accountContextMock = new Mock<AccountsContext>(_dbConnectionMock.Object);

_accountContextMock.Setup(x => x.SaveChangesAsync()).Returns(Task.FromResult(0));
_dataScopeMock = new Mock<IDataScope>();
_dataScopeMock.Setup(x => x.Connection).Returns(_dbConnectionMock.Object);

AccountServiceImp.ScopeFactory = request => _dataScopeMock.Object;
AccountServiceImp.ContextFactory = context => _accountContextMock.Object;
//Do nothing here
AccountServiceImp.AccountAddOrUpdateFactory = (set, account) => { };

_settingServiceMock = new Mock<ISettingsServiceContract>();
_settingServiceMock
.Setup(x => x.GetItemsAsync(It.IsAny<GetItemRequest>()))
.Returns(() => Task.FromResult(new ItemResponse()));
AccountServiceImp.SettingServiceFactory = () => _settingServiceMock.Object;

PermissionHelpers.SetPrincipalWithSpecificClaims(new List<Claim>
{
new Claim(AlkamiClaimTypes.AccountPermission + accountIdentifiers[0], "3"),
new Claim(AlkamiClaimTypes.AccountPermission + accountIdentifiers[1], "3"),
new Claim(AlkamiClaimTypes.AccountPermission + accountIdentifiers[2], "3")
});
}

// Decorate the unit test method(s) with the [Test] annotation or the test will not run.
// There are various options that can be used with the annotation.
[Test]
public async Task AddOrUpdateAccountAsync_Returns_Saved_Accounts()
{
// Create any objects that will be used in the request and create the request
var account1 = new Account();
var account2 = new Account();
var accounts = new List<Account> { account1, account2 };

// Create the request object
var request = new AddOrUpdateAccountRequest { ItemList = accounts };

// Complete any other setup needed for the test
var dbSetAccountMock = _accountContextMock.CreateFakeDbSet(accounts);
_accountContextMock.Object.Accounts = dbSetAccountMock.Object;

// Make the unit test call to the service.
PermissionHelpers.InjectClaimsIntoRequest(request);
var result = await _accountServiceImpl.AddOrUpdateAccountAsync(request);

// Validate the results for errors.
result.HasError.Should().BeFalse();
result.ValidationResults.Count.Should().Be(0);

// Validate that the result contains the required objects.
Assert.That(result.Accounts.Contains(account1));
Assert.That(result.Accounts.Contains(account2));
}
}
}
```

#### Service Integration Testing:

Create a new class for the integration tests and use example code below as a guide. The code file is also attached to this document.

```
using Alkami.Broker.Base;
using Alkami.Data.Validations;
using Alkami.MicroServices.Accounts.Contracts;

using Alkami.MicroServices.Accounts.Contracts.Sorters;

using Alkami.MicroServices.Accounts.Service.Client;
using Alkami.MicroServices.Accounts.Service.Host;
using Alkami.MicroServices.Accounts.Service.IntegrationTests.Builders;
using Alkami.MicroServices.Settings.Contracts;
using Alkami.MicroServices.Settings.Contracts.Requests;
using Alkami.MicroServices.Settings.Contracts.Responses;
using Alkami.Security;
using Alkami.Test;
using FluentAssertions;
using log4net.Config;
using Moq;
using NUnit.Framework;
using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;
using System.IO;
using System.Linq;
using System.Security.Claims;
using System.Threading;
using System.Threading.Tasks;

#pragma warning disable 1591

namespace USBFI.MicroServices.Accounts.Service.IntegrationTests
{
// The [TestFixture] annotation is used to inform NUnit that this class contains tests.
[TestFixture]
public class AccountsIntegrationTests : DatabaseBasedTests
{
private readonly IAccountServiceContract _testClass;
private readonly DistributedAccountService _host;
private readonly Mock<ISettingsServiceContract> _settingsServiceContract;
private AccountBuilder _accountBuilder;

public AccountsIntegrationTests()
{
MockResolver.Use();

var filepath = Path.Combine(TestContext.CurrentContext.TestDirectory, "log4net.config");
var file = new FileInfo(filepath);

if (file.Exists)
{
XmlConfigurator.Configure(file);
}

_settingsServiceContract = new Mock<ISettingsServiceContract>();
_settingsServiceContract
.Setup(x => x.GetItemsAsync(It.IsAny<GetItemRequest>()))
.Returns(() => Task.FromResult(new ItemResponse()));

AccountServiceImp.SettingServiceFactory = () => _settingsServiceContract.Object;

Alkami.Broker.App.Subscription.SubscriberFactory = () => new EmptySubscriber(CancellationToken.None);

_host = new DistributedAccountService();
_host.OnStart();

_testClass = new AccountServiceClient();
}

/// <summary>
/// This is to be run before every integration test; copy the "tester" database so that the original
/// database remains un-modified
/// </summary>
[SetUp]
public void Setup()
{
var file = new FileInfo(TestContext.CurrentContext.TestDirectory + "\\log4net.config");
if (file.Exists) { XmlConfigurator.Configure(file); }

_accountBuilder = new AccountBuilder();

AccountServiceImp.ScopeFactory = request => CreateMockedDataScope();
}

// This is used to cleanup any artifacts used for the integration tests.
// This will be run after all tests have finished running.
[OneTimeTearDown]
public void TearDown()
{

_host.OnStop(TimeSpan.FromSeconds(10));
}

// Decorate the integration test method with the [Test] annotation or the test will not run.
// There are various options that can be used with the annotation.
[Test]
public async Task AccountsMicroservice_GetAccountsAsync_ReturnsAccounts()
{
// Create any objects that will be used in the request and create the request
var maxResults = 9;
var accountsRequest = new GetAccountRequest
{
Mapping =
{
IncludeAccountType = true,
IncludeRoutingInfo = true
},
Filter =
{
IncludeAggregate = true,
IncludeExternal = true
},
MaxResults = maxResults,
};

// Complete any other setup needed for the test
SetupFixture.AugmentBaseRequest(accountsRequest, SetupFixture.Defaults.Users[1]);

// Make the integration test call to the service.
var accountResponse = await _testClass.GetAccountAsync(accountsRequest);

// Validate that the results.
Assert.IsNotNull(accountResponse);
Assert.IsFalse(accountResponse.HasError);
Assert.IsNotNull(accountResponse.Accounts);
Assert.AreEqual(maxResults, accountResponse.Accounts.Count);

foreach (var account in accountResponse.Accounts)
{
List<ValidationResult> results;
Assert.IsTrue(account.Validate(out results));
}
}
}
}
```
