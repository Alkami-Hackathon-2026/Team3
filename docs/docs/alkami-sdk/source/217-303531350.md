To provide a more in depth example of how to control the titlebar, here is an example from one of our Vue applications.

```
import Vue from 'vue';
import { INavBar } from '../interfaces/INavBar';
import { NavAction } from '../interfaces/NavAction';

export default Vue.extend({
name: 'titlebar',
data() {
return {
firstUpdate: true, // Use the setNavBar() if first update, otherwise use updateNavBar()
isNative: window.nativeHook.isNativeApp(),
};
},
props: {
leftButton: {
type: Object as () => NavAction,
default: null,
},
title: {
type: String,
default: '',
},
rightButton: {
type: Object as () => NavAction,
default: null,
},
transparent: {
type: Boolean,
default: false,
},
},
methods: {
updateTitleBar() {
// If we are in the native application, use the nativeHook methods
if (this.isNative) {
const updateNavBar = this.firstUpdate ?
(window as any).nativeHook.setNavBar :
(window as any).nativeHook.updateNavBar;

updateNavBar({
left: this.leftButton,
right: this.rightButton,
text: this.title,
});

this.firstUpdate = false;
return;
}

// Otherwise, work with the HTML titlebar through JS
const titleBarElement = document.querySelector('.titlebar');

if (!titleBarElement) {
return;
}

const buttonElements = titleBarElement.querySelectorAll('.titlebar-button');
const pageTitleElement = document.getElementById('page_title');
const leftButtonElement = titleBarElement.querySelector('.titlebar-button--left') as HTMLButtonElement;
const rightButtonElement = titleBarElement.querySelector('.titlebar-button--right') as HTMLButtonElement;

if (pageTitleElement) {
pageTitleElement.innerText = this.title || '';
}

(Array.from(buttonElements) as HTMLElement[]).forEach((element: HTMLElement) => {
element.classList.add('hidden'); // Hide all current buttons in the titlebar
});

if (this.leftButton) {
leftButtonElement.innerText = '';

switch (this.leftButton.button) {
case NavAction.drawer:
const navButton = document.getElementById('nav_button');

if (navButton) {
navButton.classList.remove('hidden');
}

break;

case NavAction.back:
const backButtonElement = document.getElementById('btnBack');

if (backButtonElement) {
backButtonElement.setAttribute('href', 'javascript:' + this.leftButton.callback);
backButtonElement.classList.remove('hidden');
}

break;

case NavAction.custom:
leftButtonElement.innerText = this.leftButton.button.text;
leftButtonElement.setAttribute('href', 'javascript:' + this.leftButton.callback);
leftButtonElement.classList.remove('hidden');
break;
}
}

if (this.rightButton) {
Array.from(rightButtonElement.classList).forEach((className) => {
if (className.startsWith('font-icon-')) {
rightButtonElement.classList.remove(className);
}
});

rightButtonElement.innerText = '';

switch (this.rightButton.button) {
case NavAction.close:
rightButtonElement.classList.add('font-icon-cancel-x');
break;

case NavAction.custom:
rightButtonElement.innerText = this.rightButton.text;
break;
}

rightButtonElement.setAttribute('href', 'javascript:' + this.rightButton.callback);
rightButtonElement.classList.remove('hidden');
}
},
},
mounted() {
const leftButtonElement = document.createElement('A');
leftButtonElement.setAttribute('class', 'titlebar-button left hidden titlebar-button--left');

const rightButtonElement = document.createElement('A');
rightButtonElement.setAttribute('class', 'titlebar-button right hidden titlebar-button--right');

const titleBarElement = document.querySelector('.titlebar') as any;
titleBarElement.prepend(leftButtonElement);
titleBarElement.append(rightButtonElement);

this.updateTitleBar();
},
watch: {
leftButton: 'updateTitleBar',
rightButton: 'updateTitleBar',
title: 'updateTitleBar',
transparent: 'updateTitleBar',
},
});
```
