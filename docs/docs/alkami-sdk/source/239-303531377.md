It's easy to test components that render differently based on a few props, but what about when you need to test some level of interaction? For this we still use the same mounting and mocking mechanisms seen in the previous example, but instead of immediately asserting a case, we trigger a user event on an element in the component.

#### Source File
RepositoryRecord.vue

```
<template>
<div class="repository">
<div
class="row repository__row"
:class="this.repositoryExecutions.length > 0 ? 'repository__row--has-executions' : ''"
@click="expandRow">

<div class="col-sm-6 repository__info-row">
<repository-info
:repositoryInfo="repositoryInfo"></repository-info>
</div>

<div class="col-sm-6 repository__row-info">
<div
v-if="codebaseStats && codebaseStats.crappyMethodCount > 0"
class="repository__warning"
data-tooltip>
<svg viewBox="0 0 32 32" xmlns="http://www.w3.org/2000/svg">...</svg>
<div class="tooltip">
<div class="tooltip__title">Warning</div>
<div class="tooltip__content">This repository has CRAPpy methods present in one of its nodes.</div>
</div>
</div>

<codebase-coverage
:coverageStats="firstExecution"></codebase-coverage>

<div
class="repository__view-more"
:class="this.repositoryExecutions.length > 0 ? '' : 'repository__view-more--disabled'"
@click.prevent.stop="goToRepositoryPage">

<svg class="repository__view-icon" viewBox="0 0 48 48" xmlns="http://www.w3.org/2000/svg">...</svg>
</div>
</div>
</div>

<div class="row repository__details" v-if="isExpanded">
<div class="col-sm-8">
<line-chart :height="350" :chartData="lineChartData"></line-chart>
</div>

<div class="col-sm-4">
<codebase-statistics
:codebaseStats="codebaseStats"></codebase-statistics>
</div>
</div>
</div>
</template>

<script lang="ts">
import Vue from 'vue';

// Imports
import { IProject } from '../../interfaces/IProject';
import { IExecution } from '../../interfaces/IExecutions';
import { IChartData, generateLineChartData } from '../../../utils/chartData';

export default Vue.extend({
name: 'repository-record',
data: () => {
return {
isExpanded: false,
};
},
props: {
repositoryInfo: {
type: Object as () => IProject,
required: true,
},
},
computed: {
codebaseStats(): IExecution['stats']['codebase'] | null {
if (this.firstExecution) {
return this.firstExecution.stats.codebase;
}

return null;
},
firstExecution(): IExecution | null {
return this.repositoryExecutions[0] || null;
},
lineChartData(): IChartData {
return generateLineChartData(this.repositoryExecutions);
},
repositoryExecutions(): IExecution[] {
return this.$store.state.executionAggregate[this.repositoryInfo.id];
},
},
methods: {
expandRow() {
if (this.repositoryExecutions.length > 0) {
this.isExpanded = !this.isExpanded;
}
},
goToRepositoryPage() {
if (this.repositoryExecutions.length > 0) {
this.$router.push(`/repository/${this.$options.filters!.formatProjectName(this.repositoryInfo.name)}`);
}
},
},
components: {
CodebaseCoverage: () => import('@/components/shared/CodebaseCoverage.vue'),
CodebaseStatistics: () => import('@/components/shared/CodebaseStatistics.vue'),
LineChart: () => import('@/components/charts/LineChart.vue'),
RepositoryInfo: () => import('@/components/project/RepositoryInfo.vue'),
},
});
</script>
```

#### Test File
RepositoryRecord.spec.ts

```
import { shallowMount } from '@vue/test-utils';
import { projectRefs } from '../../data/projectRefs';
import { executions } from '../../data/executions';
import RepositoryRecord from '../../../Scripts/components/project/RepositoryRecord.vue';

// Mock data
const repositoryInfo = projectRefs.projects[0];
const storeMock = {
$store: {
state: {
executionAggregate: {
'31DAE330A77807FA058B7867E3BD02307539FF2E': [executions.executions[0]],
},
},
},
};

describe('RepositoryRecord.vue', () => {
it('expands on row click', () => {
// Mount the component with the needed mocks (props and state are mocked here)
const wrapper = shallowMount(RepositoryRecord, {
propsData: {
repositoryInfo,
},
mocks: storeMock,
filters: {
formatProjectName(input: string) { return input; },
},
});

// Find the row element and click on it
wrapper
.find('.repository__row'); // <= Find the row element
.trigger('click'); // <= Trigger a click action on it

// Attempt to find the newly revealed element and check that it exists
const expandedElement = wrapper.find('.repository__details');
expect(expandedElement.exists()).toBe(true);
});
});
```
